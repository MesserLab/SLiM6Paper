#
# RUNNER SCRIPT FOR ALL PERFORMANCE COMPARISONS
#
# this should be run with zsh (i.e., `zsh _performance_tests.txt`)
#
# this script expects production builds of SLiM 5.2 and SLiM 6.0 to be
# in the same directory, named slim52 and slim6.
#
# the runtimes go to Terminal and get copy-pasted into plot_results.R


cd "/path/to/repo/perf_tests/"


echo "starting test 1A"
for x in $(seq 100); do ./slim52 -s $x -time performance_test_1A.slim; done |& grep 'CPU time used:' | grep -o '[0-9]*\.[0-9]*' | paste -s -d ',' -
for x in $(seq 100); do ./slim6 -s $x -time performance_test_1A.slim; done |& grep 'CPU time used:' | grep -o '[0-9]*\.[0-9]*' | paste -s -d ',' -
echo "ending test 1A"
